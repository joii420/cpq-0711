const server = http.createServer((req, res) => {
  const url = new URL(req.url, 'http://x');
  const p = url.pathname;
  const q = url.searchParams;
  res.setHeader('Content-Type', 'application/json; charset=utf-8');

  const page = Number(q.get('page') ?? 0);   // 0-based
  const size = Number(q.get('size') ?? 20);
  const keyword = (q.get('keyword') ?? '').trim();

  const paged = (all) => {
    const f = keyword
      ? all.filter((r) => Object.values(r).some((v) => v != null && String(v).includes(keyword)))
      : all;
    // 🚩 故意按 0-based 切片：前端若忘了 current-1，首屏会明显取到第二页
    return { total: f.length, items: f.slice(page * size, page * size + size) };
  };

  if (p === '/api/cpq/auth/me') {
    return res.end(ok({ id: 'u1', username: 'admin', fullName: '系统管理员', role: ROLE, forceChangePassword: false }));
  }
  if (p.startsWith('/api/cpq/notifications')) {
    return res.end(ok({ count: 0, total: 0, items: [] }));
  }
  if (p === '/api/cpq/dataset/quote/customer-parts') {
    return res.end(ok(EMPTY ? { total: 0, items: [] } : paged(CUSTOMER_PARTS)));
  }
  if (p === '/api/cpq/dataset/quote/parts') {
    if (EMPTY) return res.end(ok({ total: 0, items: [] }));
    const r = paged(PARTS);
    if (NO_PROD_NO) r.items = r.items.map(({ productionNo, ...rest }) => rest); // 缺口2 未补齐
    return res.end(ok(r));
  }
  if (p === '/api/cpq/dataset/quote/sheets') {
    return res.end(ok({ sheets: SHEETS }));
  }
  let m = p.match(/^\/api\/cpq\/dataset\/quote\/parts\/([^/]+)\/overview$/);
  if (m) {
    const axis = decodeURIComponent(m[1]);
    return res.end(ok({
      axisValue: axis,
      materialName: PARTS.find((x) => x.axisValue === axis)?.materialName ?? null,
      sheets: SHEETS.map((s) => {
        const rows = rowsFor(axis, s.sheetKey);
        return { sheetKey: s.sheetKey, rowCount: rows.length,
                 versionNo: rows.length > 0 ? 1 : null,
                 lastUpdatedAt: rows.length > 0 ? '2026-09-03T10:12:00+08:00' : null,
                 source: rows.length > 0 ? 'IMPORT' : null };
      }),
    }));
  }
  m = p.match(/^\/api\/cpq\/dataset\/quote\/parts\/([^/]+)\/sheets\/([^/]+)\/rows$/);
  if (m) {
    const rows = rowsFor(decodeURIComponent(m[1]), decodeURIComponent(m[2]));
    // 🚨 故意回 readOnly:false —— 前端必须无视它，一律只读（api.md 硬约束 7 / AC-8）
    return res.end(ok({ versionNo: rows.length ? 1 : null, isLatest: true, readOnly: false,
                        source: 'IMPORT', rows }));
  }
  m = p.match(/^\/api\/cpq\/dataset\/quote\/parts\/([^/]+)\/sheets\/([^/]+)\/versions$/);
  if (m) {
    const rows = rowsFor(decodeURIComponent(m[1]), decodeURIComponent(m[2]));
    return res.end(ok({ versions: rows.length
      ? [{ versionNo: 1, isLatest: true, rowCount: rows.length,
           updatedAt: '2026-09-03T10:12:00+08:00', updatedBy: 'admin', source: 'IMPORT' }]
      : [] }));
  }
  res.statusCode = 404;
  res.end(JSON.stringify({ code: 404, message: `mock: no route ${p}`, data: null }));
});
server.listen(PORT, () => console.log(`mock backend on ${PORT} role=${ROLE} empty=${EMPTY}`));
