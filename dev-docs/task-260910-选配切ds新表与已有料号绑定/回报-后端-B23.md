**核价侧 no-op 的量化证据**（比跑一遍核价 E2E 更强，因为它证明的是「结构上不可能变」）：

```sql

  select production_no, component_no, count(*) c from v_ds_cost_basic_material_bom_all
  where is_current and component_no is not null group by 1,2 having count(*)>1) t;
 →  dup_edge_keys=0 | rows_involved=0

-- COSTING 骨架源表
select count(*) dup_edge_keys, coalesce(sum(c),0) rows_involved from (
  select material_no, component_no, count(*) c from material_bom_item
  where customer_no='_GLOBAL_' and system_type='PRICING' and is_current and component_no is not null
  group by 1,2 having count(*)>1) t;

```
