SELECT dqm.material_no AS hf_part_no, dqm.material_no AS "_物料_销售料号", dqm.material_name AS "_物料_品名", dqm.unit_weight AS "_物料_单重", dqcp.customer_product_no AS "_客户料号_客户产品号", dqcp.customer_drawing_no AS "_客户料号_客户图号" FROM ds_quote_material dqm LEFT JOIN ds_quote_customer_part dqcp ON dqcp.material_no = dqm.material_no WHERE dqm.material_no = ANY(:total_material_no) ORDER BY dqm.material_no
